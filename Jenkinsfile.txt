pipeline {
    agent any

    stages {
        stage('Build') {
            steps {
                echo 'Building Calculator Project'
                bat 'mvn clean package'
            }
        }

        stage('Test') {
            steps {
                echo 'Running JUnit Tests'
                bat 'mvn test'
            }
        }
    }

    post {
        success {
            echo 'Build Successful!'
        }

        failure {
            echo 'Build Failed!'
        }
    }
}
